cmake_minimum_required(VERSION 3.15)
project(greenloop_route LANGUAGES CXX)
add_executable(greenloop-route route.cpp)
target_compile_features(greenloop-route PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(greenloop-route PRIVATE /utf-8 /W4)
else()
  target_compile_options(greenloop-route PRIVATE -Wall -Wextra -Wpedantic)
endif()
